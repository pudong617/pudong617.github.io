def seeded_random(seed):
    “””实现与原代码相同的线性同余生成器(LCG)”””
    # 原代码中的参数: 乘数=1664525, 增量=1013904223, 模=4751297606875873280
    multiplier = 1664525
    increment = 1013904223
    modulus = 4751297606875873280

        seed = (seed * multiplier + increment) % modulus

def unshuffle_string(shuffled_str, seed=42):
    “””还原被permuteString方法洗牌的字符串”””
    # 将字符串转换为列表以便进行交换操作
    chars = list(shuffled_str)
    length = len(chars)
    
    # 初始化随机数生成器，与原算法使用相同的种子
    rng = seeded_random(seed)
    
    # 记录洗牌时使用的随机索引
    swap_indices = []
    temp_length = length
    for _ in range(length):
        # 生成与洗牌时相同的随机索引
        rand = next(rng)
        idx = int(rand * temp_length)
        swap_indices.append(idx)
        temp_length -= 1
    
    # 按照与洗牌相反的顺序执行交换，还原原始字符串
    for i in range(length):
        # 计算当前需要处理的位置
        pos = i
        # 获取洗牌时使用的随机索引
        idx = swap_indices[length - 1 - i]
        # 执行交换
        chars[pos], chars[idx] = chars[idx], chars[pos]
    
    return’’.join(chars)
