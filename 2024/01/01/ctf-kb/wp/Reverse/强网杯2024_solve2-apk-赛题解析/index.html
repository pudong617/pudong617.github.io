((num << 8) & 0x00FF0000) | // 取第三字节
((num << 24) & 0xFF000000); // 取最低字节
}

    private static final int RS_MDS_Encode( int k0, int k1) {
    int r = k1;
    for (int i = 0; i < 4; i++) // shift 1 byte at a time
    r = RS_rem( r );
    r ^= k0;
    for (int i = 0; i < 4; i++)
    r = RS_rem( r );
    return r;
    }
