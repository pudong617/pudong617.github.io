青龙

crypto001

from Crypto.Util.number import *
from secret import flag

p = getPrime(512)
q = getPrime(512)
n = p * q
d = getPrime(299)
e = inverse(d,(p-1)*(q-1))
m = bytes_to_long(flag)
c = pow(m,e,n)
hint1 = p >> (512-70)
hint2 = q >> (512-70)

print(f”n = {n}”)
print(f”e = {e}”)
print(f”c = {c}”)
print(f”hint1 = {hint1}”)
print(f”hint2 = {hint2}”)
